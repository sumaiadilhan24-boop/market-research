## 2. PostgreSQL DDL Schema (`supabase/schema.sql`)

```sql
-- Enable UUID extension if needed
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Products Table
CREATE TABLE IF NOT EXISTS products (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    breed_or_type TEXT,
    target_price NUMERIC(10, 2) NOT NULL,
    min_acceptable_price NUMERIC(10, 2),
    icon_type TEXT DEFAULT 'emoji', -- 'emoji' | 'lucide' | 'url'
    icon_value TEXT DEFAULT '🐱',
    status TEXT DEFAULT 'AVAILABLE', -- 'AVAILABLE' | 'RESERVED' | 'SOLD'
    sold_price NUMERIC(10, 2),
    sold_at TIMESTAMPTZ,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Visits Table (Date-based market research sessions)
CREATE TABLE IF NOT EXISTS visits (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    visit_date DATE NOT NULL DEFAULT CURRENT_DATE,
    title TEXT NOT NULL,
    location TEXT,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Customer Interactions Table
CREATE TABLE IF NOT EXISTS interactions (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    visit_id BIGINT NOT NULL REFERENCES visits(id) ON DELETE CASCADE,
    product_id BIGINT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    questions JSONB DEFAULT '[]'::jsonb,
    custom_question TEXT,
    offered_price NUMERIC(10, 2),
    customer_sentiment TEXT, -- 'HIGH_INTEREST' | 'CASUAL' | 'BARGAIN' | 'READY_TO_BUY'
    customer_notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable Row Level Security (RLS) with open anon access for simple field usage
ALTER TABLE products ENABLE ROW LEVEL SECURITY;
ALTER TABLE visits ENABLE ROW LEVEL SECURITY;
ALTER TABLE interactions ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read on products" ON products FOR SELECT USING (true);
CREATE POLICY "Allow public insert on products" ON products FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update on products" ON products FOR UPDATE USING (true);
CREATE POLICY "Allow public delete on products" ON products FOR DELETE USING (true);

CREATE POLICY "Allow public read on visits" ON visits FOR SELECT USING (true);
CREATE POLICY "Allow public insert on visits" ON visits FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update on visits" ON visits FOR UPDATE USING (true);
CREATE POLICY "Allow public delete on visits" ON visits FOR DELETE USING (true);

CREATE POLICY "Allow public read on interactions" ON interactions FOR SELECT USING (true);
CREATE POLICY "Allow public insert on interactions" ON interactions FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update on interactions" ON interactions FOR UPDATE USING (true);
CREATE POLICY "Allow public delete on interactions" ON interactions FOR DELETE USING (true);
```
